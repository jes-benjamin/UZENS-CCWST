/* ============================================================================
   i18n.txt —— 自动翻译 API 配置（可选功能）
   ============================================================================
   ★ 用途：如果你【不想自己逐条填写四种语言】的文案，只填好中文（zh），
     开启本功能后，页面会自动调用翻译 API，把中文翻译成 en/tw/ja 并填回去。
   ★ 默认 enabled: '#' = 关闭，网站完全按各 pageN.txt 里写好的四语言显示，
     不发任何网络请求。想用时再把 enabled 改成 true 并填一家密钥。
   ★ 三选一即可：google（谷歌）/ bing（微软必应）/ youdao（有道），
     在 provider 里指定用哪家，只填那一家的密钥即可。
   ★ 只改引号 ' ' 里的内容；不要删逗号 , 引号 ' 花括号 {} 方括号 []。
   ============================================================================

   ---------------------- 三家 API 申请教程 ----------------------
   【谷歌 Google · Cloud Translation】
     1. 打开 https://console.cloud.google.com/ 用谷歌账号登录。
     2. 顶部“选择项目”→ 新建项目（名字随意），选中它。
     3. 左侧菜单“API 和服务 → 库”，搜索 “Cloud Translation”，点“启用”。
     4. “API 和服务 → 凭据 → 创建凭据 → API 密钥”，复制得到的密钥。
     5. 建议点开该密钥 →“应用限制”选“HTTP 引用来源(网站)”，
        填入你的域名（如 https://你的域名/*），防止密钥被盗用。
     6. 把密钥填到下面 google.apiKey。计费：每月约 50 万字符免费额度。

   【微软必应 Bing · Azure AI 翻译】
     1. 打开 https://portal.azure.com/ 用微软账号登录。
     2. 点“创建资源”→ 搜索“Translator / 翻译工具”→ 创建。
     3. 定价层选 F0（免费，每月约 200 万字符）；区域选离你近的即可。
     4. 创建完成后进入资源 →“密钥和终结点”，复制：
        · 密钥（Key 1 或 Key 2）→ 填到下面 bing.apiKey
        · 位置/区域（如 westeurope、eastasia）→ 填到下面 bing.region
     5. 若创建的是“多服务资源”必须填 region；全球资源可留空 region。

   【有道 Youdao · 文本翻译】
     1. 打开 https://ai.youdao.com/ 注册并登录（支持网易账号）。
     2. 进入“应用管理 → 创建应用”，应用类型选“文本翻译”。
     3. 在“服务列表”里为应用开通“文本翻译”服务（有免费试用额度）。
     4. 回到应用详情，复制：
        · 应用ID（appKey）→ 填到下面 youdao.appKey
        · 应用密钥（appSecret）→ 填到下面 youdao.appSecret
     5. 注意：有道是服务端 API，很多环境用浏览器直连会被跨域(CORS)拦截；
        若开启后有道翻译失败，请改用谷歌或必应，或自行加后端代理。

   ---------------------------------------------------------------------------
   ⚠ 安全提醒：任何写在前端 txt 里的 API 密钥，访问者都能在网页源码里看到。
     请务必在服务商后台给密钥设置“域名/引用来源白名单”和用量上限。
     生产站点建议用后端代理转发来彻底隐藏密钥（见 README 说明）。
   ============================================================================ */
window.SITE_PAGES = window.SITE_PAGES || {};
window.SITE_PAGES.i18nConfig = {

    /* 是否开启自动翻译：'#' = 关闭（默认）；true = 开启 */
    enabled: '#',

    /* 使用哪家：'google'（谷歌） / 'bing'（必应） / 'youdao'（有道） */
    provider: 'google',

    /* 要自动生成哪些语言（对应页面语言切换项）；中文 zh 是源语言，不用填 */
    targets: ['en', 'tw', 'ja'],

    /* 是否只翻译"还是占位/为空"的文案：true = 已有译文不覆盖（推荐）；false = 全部覆盖 */
    onlyFillMissing: true,

    /* ------ 谷歌密钥（provider = 'google' 时填这里）------ */
    google: {
        apiKey: ''
    },

    /* ------ 必应密钥（provider = 'bing' 时填这里）------ */
    bing: {
        apiKey: '',
        region: ''     // 区域，如 westeurope / eastasia；全球资源可留空
    },

    /* ------ 有道密钥（provider = 'youdao' 时填这里）------ */
    youdao: {
        appKey: '',
        appSecret: ''
    }
};
